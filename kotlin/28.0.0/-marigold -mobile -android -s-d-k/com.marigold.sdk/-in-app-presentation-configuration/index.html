InAppPresentationConfiguration

data class InAppPresentationConfiguration(var presentationStyle: InAppPresentationStyle = InAppPresentationStyle.AUTOMATIC, var bannerPosition: InAppBannerPosition = InAppBannerPosition.AUTOMATIC, var linkBehavior: InAppLinkBehavior = InAppLinkBehavior.AUTOMATIC, var linkButtonTitle: String? = null, var backgroundColor: InAppPresentationColor? = null, var titleColor: InAppPresentationColor? = null, var bodyTextColor: InAppPresentationColor? = null, var closeButtonColor: InAppPresentationColor? = null, var backdropColor: InAppPresentationColor? = null, var backdropOpacity: Double? = null, var titleFont: InAppPresentationFont? = null, var bodyFont: InAppPresentationFont? = null, var cornerRadius: Double? = null, var outerSpacing: Double? = null, var imageContentMode: InAppImageContentMode = InAppImageContentMode.AUTOMATIC, var showsImage: Boolean? = null, var showsCloseButton: Boolean? = null, var allowsGestureDismissal: Boolean? = null, var displayDuration: Double? = null)

Native UI settings. Null and AUTOMATIC inherit. Valid fields resolve in this order: message payload, callback overrides, app defaults, SDK defaults. Dimensions are dp; displayDuration is seconds, with zero meaning manual dismissal. Full-screen settings style the native frame, preserving campaign HTML.
